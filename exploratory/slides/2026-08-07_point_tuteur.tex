\section*{A. Le capital, et son échelle}

\begin{frame}{Le capital, et ce que la contagion y ajoute}
\small
\begin{block}{Le socle, qui ne dépend d'aucun paramètre non identifiable}
Sans aucune contagion ($W=0$) : $\mathbf{5\,275}$~M\euro. C'est le \textbf{plancher
défendable} ; tout ce qui est au dessus est de la contagion, donc à borner.
\end{block}

\begin{frame}{À quelle échelle se lit ce chiffre ?}
\small
\alert{\textbf{Point capital : ces montants sont un SCR de \emph{marché}, pas d'entreprise.}}
Le calage est celui du \textbf{secteur financier mondial} ($21{,}6$ incidents \textsc{tic}
matériels par an). En ne changeant \emph{que} la fréquence d'amorce, même sévérité et même
cascade :
\vspace{0.5mm}
\begin{center}\scriptsize
\begin{tabular}{@{}l r r r@{}}
\toprule
\textbf{Échelle} & \textbf{$\lambda$/an} & \textbf{SCR $99{,}5\,\%$} & \textbf{perte moy.} \\
\midrule
Secteur financier mondial (calage actuel) & $21{,}6$ & $8\,123$~M\euro & $982$~M\euro \\
Seuil \og{}$\geq 10$ événements\fg{} & $0{,}21$ & $488$~M\euro & $9{,}7$~M\euro \\
Seuil \og{}toutes firmes\fg{} & $0{,}10$ & $229$~M\euro & $4{,}6$~M\euro \\
\textbf{$\lambda$ lu à la taille de l'entité} & $\mathbf{0{,}092}$ & $\mathbf{169}$~M\euro &
$\mathbf{3{,}6}$~M\euro \\
\bottomrule
\end{tabular}
\end{center}
\vspace{1mm}
\begin{block}{Deux seuils, et pourquoi aucun des deux n'est la bonne lecture}
Les deux lignes du milieu sont des \textbf{seaux} : celui à $0{,}21$ retient les firmes à au
moins dix événements, dont les actifs médians valent \textbf{dix-neuf fois} notre entité, et
il est défini par un filtre sur la grandeur qu'on estime. D'où $\lambda$ estimé comme une
\emph{fonction} de la taille, et lu à la bonne taille.\\[2pt]
\textbf{Le fait actuariel.} La moyenne suit $\lambda$ (facteur $270$), la \textbf{VaR ne
suit pas} (facteur $48$) : \textbf{perte unique dominante}, $98{,}8\,\%$ des années sans
aucun incident.
\end{block}
\end{frame}

\begin{frame}{Ramené à une entité, ce n'est pas le niveau qui parle}
\begin{block}{Le niveau ne conclut rien}
Notre besoin de capital cyber d'entité est \textbf{borné par} $\mathbf{169}$~M\euro, contre
$\mathbf{450}$~M\euro{} de charge opérationnelle en Formule Standard, soit $0{,}38$ fois. Mais le
forfait couvre \emph{tout} l'opérationnel, nous seulement le cyber : comparer les niveaux n'a
pas de sens.
\end{block}
\begin{block}{Ce qui parle, c'est l'insensibilité du forfait}
Le forfait vaut $450$~M\euro{} pour un profil exemplaire \textbf{comme} pour un profil
défaillant. Il ne bouge pas d'un euro. Notre modèle, lui, écarte les deux de $116$ à
$169$~M\euro, soit $+45\,\%$.
\end{block}
\vspace{1mm}
{\footnotesize Le $169$ est une \textbf{borne supérieure} d'ordre de grandeur : la sévérité est
calibrée sur de grandes institutions financières et ne se transpose pas vers le bas. L'écart
entre états, lui, est un rapport et ne dépend pas de cette réserve.}
\end{frame}

\section*{B. Les vérifications demandées}

\begin{frame}{Jacobs : vérifié, la réponse tient en trois points}
\footnotesize
\textbf{1.} Ce n'est \emph{pas} un lien fréquence-sévérité, mais une \textbf{conversion de
sévérité seule} : $\ln L = 7{,}68+0{,}76\ln X+\varepsilon$ transforme un nombre
d'enregistrements en montant.
\textbf{2.} Le \textbf{capital principal n'en dépend pas} : il est calculé sur OpRisk, déjà
en dollars. PRC n'est qu'un axe de sensibilité.
\begin{center}
\includegraphics[width=0.82\linewidth]{S16_prc_jacobs_residu.png}
\end{center}
\vspace{-1mm}
\begin{block}{3. Sur la branche PRC, la dépendance est réelle mais bénigne}
\footnotesize La conversion était traitée comme \emph{exacte}, ce qui écrasait la
dispersion. En réinjectant le résidu ($\sigma_\varepsilon=0{,}523$) : le niveau monte de $\mathbf{+7\,\%}$ et la queue s'épaissit ($\xi : 1{,}03 \to 1{,}07$), mais la \textbf{largeur \emph{relative} de l'intervalle ne bouge pas} ($1{,}43 \to 1{,}41$).
\end{block}
\end{frame}

% =====================================================================
\begin{frame}{La fréquence : ce qui était posé, maintenant mesuré}
\footnotesize
$N_j \mid Y \sim \mathcal{P}(\lambda_j m)$ avec $m=\exp(aY-a^2/2)$ : le taux $\lambda$ est
une propriété d'\textbf{entité}, la charge $a$ est un choc \textbf{commun}. Ils \textbf{ne
se lisent pas au même endroit}.
\vspace{1mm}
\begin{block}{Les mélanger serait une faute}
La surdispersion d'un panel firme-année est dominée par l'\textbf{hétérogénéité de taille}
des firmes, pas par le choc commun, qui ne se lit qu'au niveau \textbf{agrégé} et
\textbf{après retrait de tendance}.
\end{block}
\begin{block}{Ce que la confrontation donne}
\textbf{Panel} ($14\,944$ obs.) : Poisson rejeté, $p=1{,}6\times10^{-30}$. Taux de
$0{,}099$ toutes firmes à $\mathbf{0{,}210}$ sur les grandes.\\[2pt]
\textbf{Série agrégée détrendée} : $a=0{,}12$ à $0{,}30$ contre $\mathbf{0{,}60}$ posé, au
dessus même de la \textbf{borne haute} de l'intervalle ($0{,}223$).
\end{block}
\begin{block}{Et la \og{}loi d'Erlang\fg{}}
Gamma de \textbf{forme $0{,}63$} ajustée, or une Erlang exige une forme \textbf{entière
$\ge 1$}. Le mélange Gamma est le bon objet, l'Erlang n'en est qu'un cas particulier.
\emph{Réponse au point bloquant de Coraline.}
\end{block}
\end{frame}
