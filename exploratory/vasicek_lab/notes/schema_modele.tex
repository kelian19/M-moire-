\node[entete] (h1) {I. Les ingrédients};

\node[bloc, below=of h1] (b11)
  {\ttl{1. Trois sources, trois rôles disjoints}
   \textbf{OpRisk} $4\,329$ transitions, pertes en dollars $\to$ le \emph{niveau}.\\
   \textbf{PRC} $15\,053$ incidents en enregistrements, converti par Jacobs
   $\ln L = 7{,}68+0{,}76\ln X+\varepsilon$ $\to$ axe de \emph{sensibilité}.\\
   \textbf{7 rapports post-incident}, $17$ enchaînements $\to$ le seul \emph{ordre}.};

\node[bloc, below=of b11] (b12)
  {\ttl{2. Fréquence : une charge systémique commune}
   $Y\sim\mathcal{N}(0,1)$ tirée \emph{une fois par an}, commune aux cinq piliers.\\[2pt]
   $m=\exp\!\bigl(aY-\tfrac{a^2}{2}\bigr)$, \; $a=0{,}60$, \; $\mathbb{E}[m]=1$\\[2pt]
   $N_j \mid Y \ \sim\ \mathcal{P}\bigl(\lambda_j\, m\bigr)$, \quad
   $\lambda_{\text{tot}}=12$, \; $\lambda_j\propto \mathrm{ROOT}_j$\\[3pt]
   \emph{C'est le mélange qui crée la surdispersion~:}
   $\dfrac{\mathrm{Var}(N)}{\mathbb{E}[N]}=1+\lambda\bigl(e^{a^2}-1\bigr)=\varphi=9{,}2$};

\node[bloc, below=of b12] (b13)
  {\ttl{3. La matrice de corrélation entre piliers}
   Le facteur commun $Y$ corrèle les comptages sans qu'on ait à poser de copule~:
   \eq{
     \mathrm{Corr}_{jk}=\frac{\lambda_j\lambda_k\bigl(e^{a^2}-1\bigr)}
     {\sqrt{\bigl(\lambda_j+\lambda_j^2(e^{a^2}-1)\bigr)
            \bigl(\lambda_k+\lambda_k^2(e^{a^2}-1)\bigr)}}
   }
   avec $e^{a^2}-1=0{,}433$. \emph{Corrélation croissante en $\lambda$~: les piliers les
   plus exposés sont aussi les plus solidaires.}};

\node[bloc, below=of b13] (b14)
  {\ttl{4. Sévérité : loi recollée}
   Corps empirique sous le seuil, queue paramétrique au dessus~:
   \eq{
     \mathbb{P}(X>x)=p_u\Bigl(1+\tfrac{\xi(x-u)}{\sigma}\Bigr)^{-1/\xi}, \quad x>u
   }
   $\xi=0{,}60$ (OpRisk), $\xi=1{,}03$ (PRC), \; $u=4{,}176$~M\euro, \; $p_u=0{,}150$.\\[2pt]
   \emph{$\xi>1$ sur PRC~: la moyenne théorique n'existe pas.}};

\node[entete, right=9mm of h1] (h2) {II. Le mécanisme de propagation};

\node[bloc, below=of h2] (b21)
  {\ttl{5. De la matrice de jugement à la matrice de contagion}
   $s_j=\sum_k \mathrm{TRANS}_{jk}$ \; la \emph{puissance de propagation} du pilier $j$.
   \eq{
     W_{jk}=g\,\frac{\mathrm{TRANS}_{jk}}{\max_s s_s}, \qquad g=0{,}90
   }
   Nombre moyen de descendants directs de $j$~:
   $e_j = g\,s_j/\max_s s_s$.\\[2pt]
   Comme $\sum_k W_{jk}\le g$, on a $\rho(W)\le g<1$~: \textbf{sous-criticité garantie par
   construction}, la cascade s'éteint toujours.};

\node[bloc, below=3mm of casc] (b22)
  {\ttl{6. Le branchement, et la loi exacte de l'étendue}
   Depuis un pilier $j$ tombé, chaque cible est tirée proportionnellement à
   $\mathrm{TRANS}_{jk}/s_j$, et infectée avec probabilité $W_{jk}$.\\[2pt]
   La loi du \textbf{nombre de piliers touchés} est calculée \emph{exactement}, sans
   Monte-Carlo, par récursion sur les $2^5$ sous-ensembles~:
   \eq{
     \mathbb{P}(\mathcal{S}=\mathcal{A})=R(\mathcal{A})\!\!\prod_{j\in\mathcal{A},\,
     k\notin\mathcal{A}}\!\!(1-W_{jk})
   }
   où le produit est la probabilité qu'\emph{aucune} fuite ne sorte de $\mathcal{A}$.};

\node[bloc, below=of b22] (b23)
  {\ttl{7. Le seuil de bascule et la copule}
   \textbf{Marge} (un pilier bascule seul)~: $K_j=\Phi^{-1}(1-p_j)$, \; $K_4=1{,}033$.
   La loi de la latente n'est ici qu'une \emph{normalisation}.\\[3pt]
   \textbf{Dépendance} (combien basculent ensemble)~: on ne tranche pas, on encadre, à
   $\tau$ de Kendall égal $\tau=1-1/\theta=0{,}444$~:
   \\[2pt]
   \quad plancher gaussien \; $\rho_G=\sin(\pi\tau/2)=0{,}643$, \; $\lambda_U=0$\\
   \quad plafond de Gumbel \; $\theta=1{,}8$, \;
   $\lambda_U=2-2^{1/\theta}=0{,}53$\\[2pt]
   \emph{Co-extrême des cinq piliers~: $0{,}17 \to 0{,}39$.}};

\node[bloc, below=of b23] (b24)
  {\ttl{8. Amplification}
   $(I-W)^{-1}=I+W+W^2+W^3+\cdots$ \; converge car $\rho(W)<1$.\\[1pt]
   \emph{Le choc direct, plus ce qu'il déclenche, plus ce que cela déclenche.
   La direction n'apparaît qu'à partir de $W^2$.}};

% COLONNE 3 : LES SORTIES

\node[entete, right=9mm of h2] (h3) {III. Les sorties, et leurs limites};

\node[sortie, below=of h3] (b31)
  {\ttl{9. Agrégation, modèle collectif}
   \eq{
     L=\sum_{i=1}^{N} X_i^{\text{amplifiée}}, \qquad L=\sum_{j=1}^{5} L_j
   }
   $L_j$ est la perte des incidents \emph{amorcés} au pilier $j$. $10^5$ années simulées,
   nombres aléatoires communs entre configurations.};

\node[sortie, below=of b31] (b32)
  {\ttl{10. Capital}
   $\mathrm{SCR}=\mathrm{VaR}_{99{,}5\,\%}(L)$, imposé par Solvabilité~II.\\[2pt]
   socle sans contagion \; $5\,275$~M\euro\\
   avec contagion, ignorance totale \; $[\,6\,858\;;\;8\,697\,]$~M\euro\\[2pt]
   \emph{On rapporte aussi la TVaR, cohérente, là où la VaR ne l'est pas.}};

\node[sortie, below=of b32] (b33)
  {\ttl{11. Allocation aux cinq piliers}
   \textbf{Euler}, pour le \emph{niveau} de capital, version TVaR~:
   \eq{
     C_j=\mathbb{E}\bigl[L_j \,\big|\, L\ge \mathrm{VaR}_{99{,}5\,\%}\bigr],
     \qquad \textstyle\sum_j C_j=\mathrm{SCR}
   }
   \textbf{Shapley}, pour le \emph{surcoût} $\Delta_{\text{DORA}}$, avec
   $v(\mathcal{A})=\mathrm{SCR}(\mathcal{A}\ \text{non conformes})-\mathrm{SCR}(\text{tous
   conformes})$~:
   \eq{
     \varphi_j=\!\!\sum_{\mathcal{A}\not\ni j}\!\!
     \frac{|\mathcal{A}|!\,(5-|\mathcal{A}|-1)!}{5!}
     \bigl[v(\mathcal{A}\cup\{j\})-v(\mathcal{A})\bigr]
   }
   \emph{La cascade est super-additive~: les effets marginaux ne somment pas au total.
   Shapley redistribue exactement l'interaction laissée de côté.}};

\node[limite, below=of b33] (b34)
  {\ttl{12. Ce qu'on ne sait pas trancher}
   $W=S+A$~: $S$ (le lien) est identifié, $A$ (le sens) ne l'est pas.\\[2pt]
   Placebo~: $z=-0{,}33$ sur les bases annuelles, $z=+3{,}93$ sur les post-mortems.\\[2pt]
   Boîte admissible $|A_{jk}|\le t\,S_{jk}$, \; $2^{10}=1024$ sommets énumérés.\\[2pt]
   $\sigma$ est \emph{constante} sur les $1024$~: elle voit $|A|$, jamais son signe.};

\node[limite, below=of b34] (b35)
  {\ttl{13. Trois étages d'incertitude}
   paramètre (bootstrap) $\times 2{,}6$ \; $\cdot$ \; dépendance $\times 1{,}8$ \; $\cdot$
   \; famille de queue $\times 5{,}5$\\[2pt]
   \textbf{Le niveau bouge dans la bande, l'ordre des piliers ne bouge pas.}};
